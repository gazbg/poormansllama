function* dfsSteps(start) {
  const visited = new Set();
  const stack = [start];
  // yield {action:'push', node:start}
  while (stack.length) {
    const node = stack[stack.length-1];
    // peek top
    if (!visited.has(node)) {
      visited.add(node);
      yield {action:'visit', node, stack: [...stack], visited:[...visited]};
      // push unvisited neighbors in reverse order
      const neighbors = adj[node].filter(n => !visited.has(n));
      // push them
      for (const n of neighbors.reverse()) {
        stack.push(n);
        yield {action:'push', node:n, stack:[...stack]};
      }
    } else {
      // top already visited, pop (backtrack)
      stack.pop();
      yield {action:'pop', node, stack:[...stack]};
    }
  }
}

Generator functions are cleaner:

function* dfsGenerator(start) {
  const color = {}, nextIdx = {}, order = [];
  const stack = [];
  for (const n of graph.nodes) { color[n.id] = 0; nextIdx[n.id] = 0; }
  const snap = (type, extra = {}) => ({
    type, stack: stack.slice(), color: {...color}, order: order.slice(),
    ...extra
  });
  const doPush = (n, from) => {
    stack.push(n); color[n] = 1; order.push(n);
    return snap('push', { node: n, from, line: 5,
      edge: from ? [from, n] : null,
      msg: from
        ? `${from} → ${n}: ${n} is unvisited, so push it and mark it discovered.`
        : `Start at ${n}: mark it discovered and push it onto the stack.` });
  };
  // initial push
  const first = doPush(start, null);
  first.line = 1;
  first.msg = `Start at ${start}: mark it discovered and push it onto the stack.`;
  yield first;
  while (stack.length) {
    const top = stack[stack.length - 1];
    const nbs = adj(top);
    const i = nextIdx[top]++;
    if (i < nbs.length) {
      const v = nbs[i];
      if (color[v] === 0) {
        yield doPush(v, top);
      } else {
        yield snap('skip', { node: v, from: top, line: 4,
          edge: [top, v],
          msg: `Edge ${top}–${v}: ${v} is already ${color[v] === 1 ? 'discovered' : 'finished'}, so skip it.` });
      }
    } else {
      stack.pop(); color[top] = 2;
      const back = stack[stack.length - 1];
      yield snap('pop', { node: top, line: 7,
        msg: back ? `${top} has no unvisited neighbors left — pop it, mark it finished, and backtrack to ${back}.`
                  : `${top} has no unvisited neighbors left — pop it and mark it finished.` });
    }
  }
  yield snap('done', { line: 8, msg: `Stack is empty — DFS complete. Visit order: ${order.join(' → ')}.` });
}

function applyStep(s) {

  clearChecking();

function randomGraph() {
  const n = 7 + Math.floor(Math.random()*4); // 7..10
  const labels = 'ABCDEFGHIJ';
  const nodes = [];
  let guard = 0;
  while (nodes.length < n && guard++ < 5000) {
    const x = 70 + Math.random() * 660;
    const y = 70 + Math.random() * 380;
    if (nodes.every(p => (p[1]-x)**2 + (p[2]-y)**2 > 120**2)) nodes.push([labels[nodes.length], x|0, y|0]);
  }
  // if guard exhausted, just allow (rare)
  // edges: spanning tree + extras
  const edges = [];
  const has = (a,b) => edges.some(([x,y]) => (x===a&&y===b)||(x===b&&y===a));
  for (let i = 1; i < n; i++) {
    const j = Math.floor(Math.random()*i);
    edges.push([labels[i], labels[j]]);
  }
  const extra = Math.floor(n/2);
  let added = 0, tries = 0;
  while (added < extra && tries++ < 100) {
    const a = Math.floor(Math.random()*n), b = Math.floor(Math.random()*n);
    if (a===b || has(labels[a],labels[b])) continue;
    edges.push([labels[a], labels[b]]); added++;
  }
  return { nodes, edges };
}

function applyStep(s) {
  // clear transient edge highlight
  for (const e of Object.values(edgeEls)) e.classList.remove('checking');
  // node colors
  for (const n of graph.nodes) {
    const c = s.color[n.id];
    nodeEls[n.id].dataset.state = c===0?'unvisited':c===1?'discovered':'finished';
    nodeEls[n.id].classList.remove('current');
  }
  const top = s.stack[s.stack.length-1];
  if (top) nodeEls[top].classList.add('current');
  // edges
  if (s.edge) {
    const e = edgeEls[key(s.edge[0], s.edge[1])];
    if (e) {
      e.classList.add('checking');
      if (s.type === 'push') e.classList.add('tree');
      else if (s.type === 'skip' && !e.classList.contains('tree')) e.classList.add('back');
    }
  }
  renderStack(s.stack);
  renderOrder(s.order);
  highlightLine(s.line);
  stepMsg.textContent = s.msg;
  state.visited = s.order.length;
  updateMeta();
}

function loadGraph(data) {
  graph = {
    nodes: data.nodes.map(([id, x, y]) => ({ id, x, y })),
    edges: data.edges.map(([a, b]) => [a, b]),
    adj: {}
  };
  for (const n of graph.nodes) graph.adj[n.id] = [];
  for (const [a, b] of graph.edges) { graph.adj[a].push(b); graph.adj[b].push(a); }
  for (const id in graph.adj) graph.adj[id].sort();
  if (!graph.adj[startId]) startId = graph.nodes[0].id;
  buildSvg();
  reset();
}

function randomGraph() {
  const labels = 'ABCDEFGHIJ';
  const n = 7 + Math.floor(Math.random() * 4); // 7..10 nodes
  const pts = [];
  let guard = 0;
  while (pts.length < n && guard++ < 8000) {
    const x = Math.round(70 + Math.random() * 660);
    const y = Math.round(70 + Math.random() * 380);
    if (pts.every(p => (p[1] - x) ** 2 + (p[2] - y) ** 2 >= 118 ** 2)) {
      pts.push([labels[pts.length], x, y]);
    }
  }
  const edges = [];
  const has = (a, b) => edges.some(([x, y]) => (x === a && y === b) || (x === b && y === a));
  for (let i = 1; i < n; i++) {          // random spanning tree → connected
    edges.push([labels[i], labels[Math.floor(Math.random() * i)]]);
  }
  const extra = Math.floor(n / 2);       // a few cross edges to skip
  let added = 0, tries = 0;
  while (added < extra && tries++ < 200) {
    const a = Math.floor(Math.random() * n), b = Math.floor(Math.random() * n);
    if (a === b || has(labels[a], labels[b])) continue;
    edges.push([labels[a], labels[b]]);
    added++;
  }
  return { nodes: pts, edges };
}

function applyStep(s) {
  for (const e of Object.values(edgeEls)) e.classList.remove('checking');
  for (const n of graph.nodes) {
    const c = s.color[n.id];
    nodeEls[n.id].dataset.state = c === 0 ? 'unvisited' : c === 1 ? 'discovered' : 'finished';
    nodeEls[n.id].classList.remove('current');
  }
  const top = s.stack[s.stack.length - 1];
  if (top) nodeEls[top].classList.add('current');
  if (s.edge) {
    const e = edgeEls[key(s.edge[0], s.edge[1])];
    if (e) {
      e.classList.add('checking');
      if (s.type === 'push') e.classList.add('tree');
      else if (s.type === 'skip' && !e.classList.contains('tree')) e.classList.add('back');
    }
  }
  renderStack(s.stack);
  renderOrder(s.order);
  highlightLine(s.line);
  stepMsg.textContent = s.msg;
  visitedCount = s.order.length;
  updateMeta();
}

function step() {
  if (finished) return;
  const r = gen.next();
  if (r.done) { finished = true; stop(); return; }
  applyStep(r.value);
  if (r.value.type === 'done') { finished = true; stop(); }
}
function play() {
  if (finished) reset();
  playing = true;
  playBtn.textContent = '⏸ Pause';
  step();
  if (playing) schedule();
}
function stop() {
  playing = false;
  if (timer) { clearTimeout(timer); timer = null; }
  playBtn.textContent = '▶ Play';
}
function schedule() {
  if (timer) clearTimeout(timer);
  if (!playing) return;
  timer = setTimeout(() => { step(); if (playing) schedule(); }, speedMs());
}
