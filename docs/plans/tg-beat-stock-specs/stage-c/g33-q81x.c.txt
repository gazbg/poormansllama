// G33: host kernels for the hybrid MoE decode that compute an expert matmul the way the CUDA kernels do (ggml-cuda/mmvq.cu, quantize.cu, vecdotq.cuh), so that a request gives the same
// numbers whether an expert sits in the VRAM cache (GPU leg) or was missed (host leg). The ggml-cpu kernels quantize the activation row to q8_K (one scale per 256 values) and run
// vec_dot_q4_K_q8_K; the GPU quantizes to q8_1 (one scale per 32 values, the scale stored as fp16, q = roundf(x/d) with d = amax/127) and takes d8 per 32 values inside the dot. The two
// roundings of the activation differ on EVERY element, which is the part of the host/device difference that matters (about 0.3% per dot); the float summation order differs only in the
// last bit and is not reproduced here.
//
// Activation row ("q81x", struct-of-arrays so that one 256-value block is three contiguous loads):
//   qs[n]      int8 quants,               d[n/32]  float: the fp16-rounded scale of each 32-block (what the GPU's dot reads),
//   s16[n/16]  int16: the sum of the quants of each 16-half-block (exact; Q4_K's mins use pairs of them, Q6_K's -32 offset uses each one).
//
// Edits: this file, ggml-cpu.h (declarations), ggml-cpu/CMakeLists.txt (source list).

#include "ggml-cpu.h"

#include <math.h>

#include <immintrin.h>
#endif

static inline float q81x_h2f(ggml_fp16_t h) {
    return ggml_fp16_to_fp32(h);
}

static inline float q81x_round_fp16(float f) {
#if defined(__AVX2__)
    return _mm_cvtss_f32(_mm_cvtph_ps(_mm_cvtps_ph(_mm_set_ss(f), _MM_FROUND_TO_NEAREST_INT)));
#else
    return ggml_fp16_to_fp32(ggml_fp32_to_fp16(f));
#endif
}

// ---------------------------------------------------------------------------------------------------- quantizer

void ggml_cpu_quantize_q81x_ref(const float * x, int64_t n, int8_t * qs, float * d, int16_t * s16) {
    for (int64_t b = 0; b < n/32; ++b) {
        const float * xb = x + b*32;
        float amax = 0.0f;
        for (int i = 0; i < 32; ++i) {
            const float a = fabsf(xb[i]);
            amax = a > amax ? a : amax;
        }
        const float dd = amax/127.0f;
        int s0 = 0, s1 = 0;
        for (int i = 0; i < 32; ++i) {
            const int8_t q = amax == 0.0f ? 0 : (int8_t) roundf(xb[i]/dd);
            qs[b*32 + i] = q;
            if (i < 16) { s0 += q; } else { s1 += q; }
        }
        d[b]       = q81x_round_fp16(dd);
        s16[2*b]   = (int16_t) s0;
        s16[2*b+1] = (int16_t) s1;
    }
}

#if defined(__AVX2__)
// roundf for 8 floats: half away from zero (not the hardware's half-to-even)
static inline __m256 q81x_roundf8(__m256 v) {
    const __m256 t     = _mm256_round_ps(v, _MM_FROUND_TO_ZERO | _MM_FROUND_NO_EXC);
    const __m256 f     = _mm256_sub_ps(v, t); // exact for |v| < 2^23
    const __m256 absf  = _mm256_andnot_ps(_mm256_set1_ps(-0.0f), f);
    const __m256 ge    = _mm256_cmp_ps(absf, _mm256_set1_ps(0.5f), _CMP_GE_OQ);
    const __m256 sgn   = _mm256_or_ps(_mm256_and_ps(v, _mm256_set1_ps(-0.0f)), _mm256_set1_ps(1.0f));
    return _mm256_blendv_ps(t, _mm256_add_ps(t, sgn), ge);
}
#endif

void ggml_cpu_quantize_q81x(const float * x, int64_t n, int8_t * qs, float * d, int16_t * s16) {
#if defined(__AVX2__)
    const __m256 absmask = _mm256_set1_ps(-0.0f);
    for (int64_t b = 0; b < n/32; ++b) {
        const float * xb = x + b*32;
        const __m256 x0 = _mm256_loadu_ps(xb);
        const __m256 x1 = _mm256_loadu_ps(xb + 8);
        const __m256 x2 = _mm256_loadu_ps(xb + 16);
        const __m256 x3 = _mm256_loadu_ps(xb + 24);
        __m256 am = _mm256_max_ps(_mm256_max_ps(_mm256_andnot_ps(absmask, x0), _mm256_andnot_ps(absmask, x1)),
                                  _mm256_max_ps(_mm256_andnot_ps(absmask, x2), _mm256_andnot_ps(absmask, x3)));
        __m128 a4 = _mm_max_ps(_mm256_castps256_ps128(am), _mm256_extractf128_ps(am, 1));
        a4 = _mm_max_ps(a4, _mm_movehl_ps(a4, a4));
        a4 = _mm_max_ss(a4, _mm_shuffle_ps(a4, a4, 1));
        const float amax = _mm_cvtss_f32(a4);
        const float dd   = amax/127.0f;
        d[b] = q81x_round_fp16(dd);
        if (amax == 0.0f) {
            memset(qs + b*32, 0, 32);
            s16[2*b] = 0;
            s16[2*b+1] = 0;
            continue;
        }
        const __m256 vd = _mm256_set1_ps(dd);
        const __m256i i0 = _mm256_cvttps_epi32(q81x_roundf8(_mm256_div_ps(x0, vd)));
        const __m256i i1 = _mm256_cvttps_epi32(q81x_roundf8(_mm256_div_ps(x1, vd)));
        const __m256i i2 = _mm256_cvttps_epi32(q81x_roundf8(_mm256_div_ps(x2, vd)));
        const __m256i i3 = _mm256_cvttps_epi32(q81x_roundf8(_mm256_div_ps(x3, vd)));
        // |q| <= 127: the saturating packs are exact. packs works per 128-bit lane, so the order is fixed up with a permute afterwards
        const __m256i p01 = _mm256_packs_epi32(i0, i1); // 0 1 | 0 1 -> lanes: [i0.lo i1.lo | i0.hi i1.hi]
        const __m256i p23 = _mm256_packs_epi32(i2, i3);
        const __m256i p   = _mm256_packs_epi16(p01, p23);
        const __m256i pf  = _mm256_permutevar8x32_epi32(p, _mm256_setr_epi32(0, 4, 1, 5, 2, 6, 3, 7));
        _mm256_storeu_si256((__m256i *) (qs + b*32), pf);
        // sums of the two 16-halves, from the stored bytes
        const __m256i w0 = _mm256_cvtepi8_epi16(_mm_loadu_si128((const __m128i *) (qs + b*32)));
        const __m256i w1 = _mm256_cvtepi8_epi16(_mm_loadu_si128((const __m128i *) (qs + b*32 + 16)));
        const __m256i one = _mm256_set1_epi16(1);
        const __m256i m0 = _mm256_madd_epi16(w0, one); // 8 x int32
        const __m256i m1 = _mm256_madd_epi16(w1, one);
        __m128i h0 = _mm_add_epi32(_mm256_castsi256_si128(m0), _mm256_extracti128_si256(m0, 1));
        __m128i h1 = _mm_add_epi32(_mm256_castsi256_si128(m1), _mm256_extracti128_si256(m1, 1));
        h0 = _mm_add_epi32(h0, _mm_shuffle_epi32(h0, 0x4E));
        h0 = _mm_add_epi32(h0, _mm_shuffle_epi32(h0, 0xB1));
        h1 = _mm_add_epi32(h1, _mm_shuffle_epi32(h1, 0x4E));
        h1 = _mm_add_epi32(h1, _mm_shuffle_epi32(h1, 0xB1));
        s16[2*b]   = (int16_t) _mm_cvtsi128_si32(h0);
        s16[2*b+1] = (int16_t) _mm_cvtsi128_si32(h1);
    }
#else
    ggml_cpu_quantize_q81x_ref(x, n, qs, d, s16);
#endif
}

// ---------------------------------------------------------------------------------------------------- Q4_K

static inline void q81x_q4K_scales(const uint8_t * s12, uint8_t * sc /* [8] */, uint8_t * mn /* [8] */) {
    // the 6-bit scales and mins of a block_q4_K (same unpacking as ggml_vec_dot_q4_K_q8_K)
    uint32_t utmp[4];
    memcpy(utmp, s12, 12);
    utmp[3] = ((utmp[2] >> 4) & 0x0f0f0f0fu) | (((utmp[1] >> 6) & 0x03030303u) << 4);
    const uint32_t uaux = utmp[1] & 0x3f3f3f3fu;
    utmp[1] = (utmp[2] & 0x0f0f0f0fu) | (((utmp[0] >> 6) & 0x03030303u) << 4);
    utmp[2] = uaux;
    utmp[0] &= 0x3f3f3f3fu;
    memcpy(sc, &utmp[0], 8);
    memcpy(mn, &utmp[2], 8);
}

void ggml_cpu_vec_dot_q4_K_q81x_ref(int n, float * s, const void * vx, const int8_t * qs, const float * d8, const int16_t * s16) {
    const block_q4_K * x = (const block_q4_K *) vx;
    const int nb = n/QK_K;
    float total = 0.0f;
    for (int i = 0; i < nb; ++i) {
        uint8_t sc[8], mn[8];
        q81x_q4K_scales(x[i].scales, sc, mn);
        const float dx   = q81x_h2f(x[i].d);
        const float dmin = q81x_h2f(x[i].dmin);
        float sumf_d = 0.0f, sumf_m = 0.0f;
        for (int j = 0; j < 8; ++j) {
            const int jj = j/2; // the 64-value group; even j = low nibbles, odd j = high nibbles
            const uint8_t * q4 = x[i].qs + 32*jj;
            const int8_t  * q8 = qs + (size_t) i*QK_K + 32*j;
            int dot = 0;
            for (int t = 0; t < 32; ++t) {
                const int v = (j & 1) ? (q4[t] >> 4) : (q4[t] & 0x0F);
                dot += v*q8[t];
            }
            const int sumq = s16[(size_t) i*16 + 2*j] + s16[(size_t) i*16 + 2*j + 1];
            const float d8j = d8[(size_t) i*8 + j];
            sumf_d += d8j*(float) (dot*sc[j]);
            sumf_m += d8j*(float) (sumq*mn[j]);
        }
        total += dx*sumf_d - dmin*sumf_m;
    }
    *s = total;
}

#if defined(__AVX2__)
static inline float q81x_hsum8(__m256 v) {
    __m128 lo = _mm256_castps256_ps128(v);
    __m128 hi = _mm256_extractf128_ps(v, 1);
    lo = _mm_add_ps(lo, hi);
    lo = _mm_add_ps(lo, _mm_movehl_ps(lo, lo));
    lo = _mm_add_ss(lo, _mm_movehdup_ps(lo));
    return _mm_cvtss_f32(lo);
}
#endif

void ggml_cpu_vec_dot_q4_K_q81x(int n, float * s, const void * vx, const int8_t * qs, const float * d8, const int16_t * s16) {
#if defined(__AVX2__)
    const block_q4_K * x = (const block_q4_K *) vx;
    const int nb = n/QK_K;
    const __m256i m4  = _mm256_set1_epi8(0xF);
    const __m256i one = _mm256_set1_epi16(1);
    __m256 acc  = _mm256_setzero_ps();
    __m256 accm = _mm256_setzero_ps();
    for (int i = 0; i < nb; ++i) {
        uint8_t sc[8], mn[8];
        q81x_q4K_scales(x[i].scales, sc, mn);
        const float dx   = q81x_h2f(x[i].d);
        const float dmin = q81x_h2f(x[i].dmin);
        const uint8_t * q4 = x[i].qs;
        const int8_t  * q8 = qs + (size_t) i*QK_K;
        const float   * db = d8 + (size_t) i*8;
        for (int jj = 0; jj < 4; ++jj) {
            const __m256i q4bits = _mm256_loadu_si256((const __m256i *) q4); q4 += 32;
            const __m256i q4l = _mm256_and_si256(q4bits, m4);
            const __m256i q4h = _mm256_and_si256(_mm256_srli_epi16(q4bits, 4), m4);
            const __m256i q8l = _mm256_loadu_si256((const __m256i *) q8); q8 += 32;
            const __m256i q8h = _mm256_loadu_si256((const __m256i *) q8); q8 += 32;
            const __m256i p32l = _mm256_madd_epi16(_mm256_set1_epi16((short) sc[2*jj]),     _mm256_maddubs_epi16(q4l, q8l));
            const __m256i p32h = _mm256_madd_epi16(_mm256_set1_epi16((short) sc[2*jj + 1]), _mm256_maddubs_epi16(q4h, q8h));
            acc = _mm256_fmadd_ps(_mm256_cvtepi32_ps(p32l), _mm256_set1_ps(dx*db[2*jj]),     acc);
            acc = _mm256_fmadd_ps(_mm256_cvtepi32_ps(p32h), _mm256_set1_ps(dx*db[2*jj + 1]), acc);
        }
        // mins: sum over the 8 sub-blocks of d8_j * m_j * (sum of the quants of sub-block j)
        const __m256i sq32 = _mm256_madd_epi16(_mm256_loadu_si256((const __m256i *) (s16 + (size_t) i*16)), one); // 8 x (s16[2j] + s16[2j+1])
        const __m256i m32  = _mm256_cvtepu8_epi32(_mm_loadl_epi64((const __m128i *) mn));
        const __m256  pf   = _mm256_cvtepi32_ps(_mm256_mullo_epi32(m32, sq32));
        accm = _mm256_fmadd_ps(_mm256_mul_ps(pf, _mm256_loadu_ps(db)), _mm256_set1_ps(dmin), accm);
    }
    *s = q81x_hsum8(acc) - q81x_hsum8(accm);
#else
    ggml_cpu_vec_dot_q4_K_q81x_ref(n, s, vx, qs, d8, s16);
#endif
}

// ---------------------------------------------------------------------------------------------------- Q6_K

void ggml_cpu_vec_dot_q6_K_q81x_ref(int n, float * s, const void * vx, const int8_t * qs, const float * d8, const int16_t * s16) {
    (void) s16;
    const block_q6_K * x = (const block_q6_K *) vx;
    const int nb = n/QK_K;
    float total = 0.0f;
    for (int i = 0; i < nb; ++i) {
        const float dx = q81x_h2f(x[i].d);
        float sumf = 0.0f;
        for (int b = 0; b < 8; ++b) {              // one q8_1 block = 32 values = one quarter of a 128-half
            const int h = b/4, q = b%4;
            const uint8_t * ql = x[i].ql + 64*h;
            const uint8_t * qh = x[i].qh + 32*h;
            const int8_t  * q8 = qs + (size_t) i*QK_K + 32*b;
            int dot[2] = {0, 0};
            for (int l = 0; l < 32; ++l) {
                int v;
                switch (q) {
                    case 0:  v = (ql[l]      & 0xF) | (((qh[l] >> 0) & 3) << 4); break;
                    case 1:  v = (ql[l + 32] & 0xF) | (((qh[l] >> 2) & 3) << 4); break;
                    case 2:  v = (ql[l]      >> 4)  | (((qh[l] >> 4) & 3) << 4); break;
                    default: v = (ql[l + 32] >> 4)  | (((qh[l] >> 6) & 3) << 4); break;
                }
                dot[l/16] += (v - 32)*q8[l];
            }
            const int sc0 = x[i].scales[8*h + 2*q], sc1 = x[i].scales[8*h + 2*q + 1];
            sumf += d8[(size_t) i*8 + b]*(float) (dot[0]*sc0 + dot[1]*sc1);
        }
        total += dx*sumf;
    }
    *s = total;
}

void ggml_cpu_vec_dot_q6_K_q81x(int n, float * s, const void * vx, const int8_t * qs, const float * d8, const int16_t * s16) {
#if defined(__AVX2__)
    const block_q6_K * x = (const block_q6_K *) vx;
    const int nb = n/QK_K;
    const __m256i m3  = _mm256_set1_epi8(3);
    const __m256i m15 = _mm256_set1_epi8(15);
    __m256 acc  = _mm256_setzero_ps();
    __m256 accm = _mm256_setzero_ps();
    for (int i = 0; i < nb; ++i) {
        const float dx = q81x_h2f(x[i].d);
        const uint8_t * ql = x[i].ql;
        const uint8_t * qh = x[i].qh;
        const int8_t  * q8 = qs + (size_t) i*QK_K;
        const float   * db = d8 + (size_t) i*8;

        const __m128i sc128 = _mm_loadu_si128((const __m128i *) x[i].scales);
        const __m256i sc16  = _mm256_cvtepi8_epi16(sc128);
        // the -32 offset: sum over each pair of 16-groups (= each q8_1 block b) of sc_g * (sum of the quants of group g), times 32, as 8 int32 lanes b = 0..7
        const __m256i bias32 = _mm256_slli_epi32(_mm256_madd_epi16(_mm256_loadu_si256((const __m256i *) (s16 + (size_t) i*16)), sc16), 5);
        const __m256  biasf  = _mm256_mul_ps(_mm256_cvtepi32_ps(bias32), _mm256_loadu_ps(db));
        accm = _mm256_fmadd_ps(biasf, _mm256_set1_ps(dx), accm);

        for (int h = 0; h < 2; ++h) {
            const __m256i qlA = _mm256_loadu_si256((const __m256i *) ql); ql += 32;
            const __m256i qlB = _mm256_loadu_si256((const __m256i *) ql); ql += 32;
            const __m256i qhv = _mm256_loadu_si256((const __m256i *) qh); qh += 32;
            const __m256i h0 = _mm256_slli_epi16(_mm256_and_si256(qhv, m3), 4);
            const __m256i h1 = _mm256_slli_epi16(_mm256_and_si256(qhv, _mm256_set1_epi8(12)), 2);
            const __m256i h2 = _mm256_and_si256(qhv, _mm256_set1_epi8(48));
            const __m256i h3 = _mm256_srli_epi16(_mm256_and_si256(qhv, _mm256_set1_epi8(-64)), 2);
            const __m256i v0 = _mm256_or_si256(_mm256_and_si256(qlA, m15), h0);
            const __m256i v1 = _mm256_or_si256(_mm256_and_si256(qlB, m15), h1);
            const __m256i v2 = _mm256_or_si256(_mm256_and_si256(_mm256_srli_epi16(qlA, 4), m15), h2);
            const __m256i v3 = _mm256_or_si256(_mm256_and_si256(_mm256_srli_epi16(qlB, 4), m15), h3);
            const __m256i vv[4] = { v0, v1, v2, v3 };
            for (int q = 0; q < 4; ++q) {
                const int b = 4*h + q;
                const __m256i y8 = _mm256_loadu_si256((const __m256i *) q8); q8 += 32;
                const __m256i p16 = _mm256_maddubs_epi16(vv[q], y8);
                // 16 int16 lanes: the first 8 belong to group 2b, the next 8 to group 2b+1
                const __m256i scv = _mm256_set_m128i(_mm_set1_epi16((short) x[i].scales[2*b + 1]), _mm_set1_epi16((short) x[i].scales[2*b]));
                const __m256i p32 = _mm256_madd_epi16(scv, p16);
                acc = _mm256_fmadd_ps(_mm256_cvtepi32_ps(p32), _mm256_set1_ps(dx*db[b]), acc);
            }
        }
    }
    *s = q81x_hsum8(acc) - q81x_hsum8(accm);
#else
    ggml_cpu_vec_dot_q6_K_q81x_ref(n, s, vx, qs, d8, s16);
#endif
}
