def patch(path, edits):
    d = open(path, "rb").read().decode("utf-8")
    crlf = "\r\n" in d
    text = d.replace("\r\n", "\n")
    for old, new in edits:
        if text.count(old) != 1:
            sys.exit("anchor matched %d times in %s: %r" % (text.count(old), path, old[:100]))
        text = text.replace(old, new)
    if os.environ.get("PATCH_DRY"):
        print("DRY-OK", path)
        return
    if os.environ.get("G22_APPLY") != "1":
        sys.exit("refusing to edit the nested tree: set G22_APPLY=1 (PATCH_DRY=1 first)")
    if crlf:
        text = text.replace("\n", "\r\n")
    open(path, "wb").write(text.encode("utf-8"))
    print("patched", path)
