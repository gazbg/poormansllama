def patch(path, edits):
    d = open(path, "rb").read().decode("utf-8")
    crlf = "\r\n" in d
    text = d.replace("\r\n", "\n")
    for old, new in edits:
        if text.count(old) != 1:
            sys.exit("anchor matched %d times in %s: %r" % (text.count(old), path, old[:80]))
        text = text.replace(old, new)
    if os.environ.get("PATCH_DRY"):
        print("DRY-OK", path)
        return
    if os.environ.get("G22_APPLY") != "1":
        sys.exit("refusing to edit the nested tree: set G22_APPLY=1 (PATCH_DRY=1 first)")
    if crlf:
        text = text.replace("\n", "\r\n")
    open(path, "wb").write(text.encode("utf-8"))
    print("patched", path)

    const int n_segs     = (n_chunks + seg_chunks - 1) / seg_chunks;
''', '''    // segments: as many whole chunks as fit in `segment` rows (the slab half has that many rows); the first chunk of a resumed pass can be short
    std::vector<int> seg_first;

            if (c == 0 || rows + pl.widths[c] > P.segment) {

                rows = 0;
            }
            rows += pl.widths[c];
        }
        seg_first.push_back(n_chunks);
    }
    const int n_segs = (int) seg_first.size() - 1;
'''),
