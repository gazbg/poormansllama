// G33: how close is the hybrid MoE host leg to the CUDA kernels it stands in for? Random Q4_K / Q6_K expert banks of the real shapes (n_embd 2048, n_ff 512) go through
//   up = mul_mat_id, gate = mul_mat_id, act = swiglu(gate, up), down = mul_mat_id
// on the CUDA backend (the graph the hybrid engine builds for the cache-resident experts) and through two host emulations:
//   old: the ggml-cpu path (activation quantized to q8_K, vec_dot_q4_K_q8_K), what the lean team did until G33,
//   new: the q8_1-style path of ggml-cpu/q81x.c (the lean team with LLAMA_MOE_HYBRID_Q81=1).
// It prints the distance of each host result to the GPU result (relative to the RMS of the GPU output) and the share of elements that are bit-equal. Needs a GPU: run it under the GPU mutex.
//
//   test-moe-q81x-cuda
#include "ggml-alloc.h"
#include "ggml-backend.h"
#include "ggml-cpu.h"
#include "ggml.h"

#include <algorithm>
#include <cmath>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <random>
#include <vector>

struct stats {
    double max_rel = 0.0, mean_rel = 0.0;
    size_t n = 0, equal = 0;
    void add(const float * a, const float * g, size_t cnt, double rms) {
        for (size_t i = 0; i < cnt; ++i) {
            const double r = std::fabs((double) a[i] - (double) g[i])/rms;
            max_rel = std::max(max_rel, r);
            mean_rel += r;
            equal += a[i] == g[i] ? 1 : 0;
        }
        n += cnt;
    }
    void print(const char * what) const {
        std::printf("  %-4s vs GPU: max %.3e, mean %.3e of the output RMS, bit-equal %.2f%%\n", what, max_rel, mean_rel/(double) std::max<size_t>(n, 1), 100.0*(double) equal/(double) std::max<size_t>(n, 1));
    }
};

struct bank_t {
    ggml_type          type;
    int64_t            k, rows, n_exp;
    std::vector<char>  data;
    size_t             row_bytes, exp_bytes;
};

static bank_t make_bank(ggml_type type, int64_t k, int64_t rows, int64_t n_exp, std::mt19937 & rng) {
    bank_t b;
    b.type = type; b.k = k; b.rows = rows; b.n_exp = n_exp;
    std::normal_distribution<float> nd(0.0f, 0.04f);
    std::vector<float> f((size_t) (k*rows*n_exp));
    for (auto & v : f) { v = nd(rng); }
    b.row_bytes = ggml_row_size(type, k);
    b.exp_bytes = b.row_bytes*(size_t) rows;
    b.data.resize(b.exp_bytes*(size_t) n_exp);
    ggml_quantize_chunk(type, f.data(), b.data.data(), 0, rows*n_exp, k, nullptr);
    return b;
}

int main() {
    ggml_backend_t gpu = nullptr;
    for (size_t i = 0; i < ggml_backend_dev_count(); ++i) {
        ggml_backend_dev_t dev = ggml_backend_dev_get(i);
        if (ggml_backend_dev_type(dev) == GGML_BACKEND_DEVICE_TYPE_GPU) {
            gpu = ggml_backend_dev_init(dev, nullptr);
            std::printf("device: %s\n", ggml_backend_dev_name(dev));
            break;
        }
    }
    if (gpu == nullptr) {
        std::printf("no GPU backend: nothing to compare\n");
        return 2;
    }

    const int64_t n_embd = 2048, n_ff = 512, n_exp = 8, n_used = 4;
    struct dcase { ggml_type up, down; };
    const dcase dcases[] = { { GGML_TYPE_Q4_K, GGML_TYPE_Q4_K }, { GGML_TYPE_Q4_K, GGML_TYPE_Q6_K } };
    std::mt19937 rng(777);
    std::normal_distribution<float> nd(0.0f, 1.0f);

    for (const dcase & dc : dcases) {
        std::printf("expert banks %s / %s / %s\n", ggml_type_name(dc.up), ggml_type_name(dc.up), ggml_type_name(dc.down));
        bank_t up = make_bank(dc.up, n_embd, n_ff, n_exp, rng), gate = make_bank(dc.up, n_embd, n_ff, n_exp, rng), down = make_bank(dc.down, n_ff, n_embd, n_exp, rng);
        for (int64_t T : { 1, 3 }) {
            stats s_old, s_new, s_new2;
            for (int rep = 0; rep < 20; ++rep) {
                std::vector<float> x((size_t) (n_embd*T));
                const float sc = std::pow(10.0f, (float) (rng() % 100)/100.0f*2.0f - 1.0f);
                for (auto & v : x) { v = nd(rng)*sc; }
                std::vector<int32_t> ids((size_t) (n_used*T));
                for (int64_t t = 0; t < T; ++t) {
                    std::vector<int> perm((size_t) n_exp);
                    for (int i = 0; i < n_exp; ++i) { perm[(size_t) i] = i; }
                    std::shuffle(perm.begin(), perm.end(), rng);
                    for (int64_t e = 0; e < n_used; ++e) { ids[(size_t) (t*n_used + e)] = perm[(size_t) e]; }
                }

                // ---- GPU: the graph of the cache-resident experts
                ggml_init_params ip = { 1u << 22, nullptr, true };
                ggml_context * ctx = ggml_init(ip);
                ggml_tensor * tu = ggml_new_tensor_3d(ctx, up.type, n_embd, n_ff, n_exp);
                ggml_tensor * tg = ggml_new_tensor_3d(ctx, gate.type, n_embd, n_ff, n_exp);
                ggml_tensor * td = ggml_new_tensor_3d(ctx, down.type, n_ff, n_embd, n_exp);
                ggml_tensor * tx = ggml_new_tensor_3d(ctx, GGML_TYPE_F32, n_embd, 1, T);
                ggml_tensor * ti = ggml_new_tensor_2d(ctx, GGML_TYPE_I32, n_used, T);
                ggml_tensor * u  = ggml_mul_mat_id(ctx, tu, tx, ti);
                ggml_tensor * g  = ggml_mul_mat_id(ctx, tg, tx, ti);
                ggml_tensor * a  = ggml_swiglu_split(ctx, g, u);
                ggml_tensor * o  = ggml_mul_mat_id(ctx, td, a, ti); // [n_embd, n_used, T]
                ggml_cgraph * gf = ggml_new_graph(ctx);
                ggml_build_forward_expand(gf, o);
                ggml_backend_buffer_t buf = ggml_backend_alloc_ctx_tensors(ctx, gpu);
                ggml_backend_tensor_set(tu, up.data.data(), 0, up.data.size());
                ggml_backend_tensor_set(tg, gate.data.data(), 0, gate.data.size());
                ggml_backend_tensor_set(td, down.data.data(), 0, down.data.size());
                ggml_backend_tensor_set(tx, x.data(), 0, x.size()*sizeof(float));
                ggml_backend_tensor_set(ti, ids.data(), 0, ids.size()*sizeof(int32_t));
                if (ggml_backend_graph_compute(gpu, gf) != GGML_STATUS_SUCCESS) {
                    std::printf("graph compute failed\n");
                    return 1;
                }
                std::vector<float> og((size_t) (n_embd*n_used*T));
                ggml_backend_tensor_get(o, og.data(), 0, og.size()*sizeof(float));
                ggml_backend_buffer_free(buf);
                ggml_free(ctx);

                double rms = 0.0;
                for (float v : og) { rms += (double) v*(double) v; }
                rms = std::sqrt(rms/(double) og.size());

                // ---- host, new: q8_1-style
                std::vector<float> hn(og.size()), ho(og.size()), hn2(og.size());
                for (int64_t t = 0; t < T; ++t) {
                    std::vector<int8_t> q1((size_t) n_embd); std::vector<float> d1((size_t) (n_embd/32)); std::vector<int16_t> s1((size_t) (n_embd/16));
                    ggml_cpu_quantize_q81x(x.data() + t*n_embd, n_embd, q1.data(), d1.data(), s1.data());
                    const ggml_type_traits_cpu * tq = ggml_get_type_traits_cpu(GGML_TYPE_Q8_K);
                    std::vector<char> xk((size_t) ggml_row_size(GGML_TYPE_Q8_K, n_embd));
                    tq->from_float(x.data() + t*n_embd, xk.data(), n_embd);
                    for (int64_t e = 0; e < n_used; ++e) {
                        const int32_t id = ids[(size_t) (t*n_used + e)];
                        std::vector<float> upv((size_t) n_ff), gav((size_t) n_ff), upo((size_t) n_ff), gao((size_t) n_ff);
                        for (int64_t r = 0; r < n_ff; ++r) {
                            ggml_cpu_vec_dot_q4_K_q81x((int) n_embd, &upv[(size_t) r], up.data.data()   + (size_t) id*up.exp_bytes   + (size_t) r*up.row_bytes,   q1.data(), d1.data(), s1.data());
                            ggml_cpu_vec_dot_q4_K_q81x((int) n_embd, &gav[(size_t) r], gate.data.data() + (size_t) id*gate.exp_bytes + (size_t) r*gate.row_bytes, q1.data(), d1.data(), s1.data());
                            ggml_get_type_traits_cpu(up.type)->vec_dot((int) n_embd, &upo[(size_t) r], 0, up.data.data()   + (size_t) id*up.exp_bytes   + (size_t) r*up.row_bytes,   0, xk.data(), 0, 1);
                            ggml_get_type_traits_cpu(up.type)->vec_dot((int) n_embd, &gao[(size_t) r], 0, gate.data.data() + (size_t) id*gate.exp_bytes + (size_t) r*gate.row_bytes, 0, xk.data(), 0, 1);
                        }
                        std::vector<float> actn((size_t) n_ff), acto((size_t) n_ff);
                        std::vector<float> actm((size_t) n_ff);
                        for (int64_t i = 0; i < n_ff; ++i) { const float gg = gav[(size_t) i]; actm[(size_t) i] = (gg/(1.0f + expf(-gg)))*upv[(size_t) i]; }
                        ggml_cpu_swiglu_f32((int) n_ff, actn.data(), gav.data(), upv.data());
                        ggml_cpu_swiglu_f32((int) n_ff, acto.data(), gao.data(), upo.data());
                        std::vector<int8_t> q2((size_t) n_ff); std::vector<float> d2((size_t) (n_ff/32)); std::vector<int16_t> s2((size_t) (n_ff/16));
                        ggml_cpu_quantize_q81x(actn.data(), n_ff, q2.data(), d2.data(), s2.data());
                        std::vector<int8_t> q3((size_t) n_ff); std::vector<float> d3((size_t) (n_ff/32)); std::vector<int16_t> s3((size_t) (n_ff/16));
                        ggml_cpu_quantize_q81x(actm.data(), n_ff, q3.data(), d3.data(), s3.data());
                        const ggml_type_traits_cpu * tdn = ggml_get_type_traits_cpu(down.type);
                        std::vector<char> ak((size_t) ggml_row_size(tdn->vec_dot_type, n_ff));
                        ggml_get_type_traits_cpu(tdn->vec_dot_type)->from_float(acto.data(), ak.data(), n_ff);
                        for (int64_t r = 0; r < n_embd; ++r) {
                            const char * wr = down.data.data() + (size_t) id*down.exp_bytes + (size_t) r*down.row_bytes;
                            float * dstn = &hn[(size_t) ((t*n_used + e)*n_embd + r)];
                            if (down.type == GGML_TYPE_Q4_K) { ggml_cpu_vec_dot_q4_K_q81x((int) n_ff, dstn, wr, q2.data(), d2.data(), s2.data()); }
                            else                              { ggml_cpu_vec_dot_q6_K_q81x((int) n_ff, dstn, wr, q2.data(), d2.data(), s2.data()); }
                            float * dstn2 = &hn2[(size_t) ((t*n_used + e)*n_embd + r)];
                            if (down.type == GGML_TYPE_Q4_K) { ggml_cpu_vec_dot_q4_K_q81x((int) n_ff, dstn2, wr, q3.data(), d3.data(), s3.data()); }
                            else                              { ggml_cpu_vec_dot_q6_K_q81x((int) n_ff, dstn2, wr, q3.data(), d3.data(), s3.data()); }
                            tdn->vec_dot((int) n_ff, &ho[(size_t) ((t*n_used + e)*n_embd + r)], 0, wr, 0, ak.data(), 0, 1);
                        }
                    }
                }
                s_new.add(hn.data(), og.data(), og.size(), rms);
                s_old.add(ho.data(), og.data(), og.size(), rms);
                s_new2.add(hn2.data(), og.data(), og.size(), rms);
            }
            std::printf(" T = %lld tokens, 20 requests of %d experts:\n", (long long) T, (int) n_used);
            s_old.print("old");
            s_new.print("new");
            s_new2.print("new+libm silu");
        }
    }
    ggml_backend_free(gpu);
    return 0;
}
