                               checkpoint rules while it fills the batch (against the checkpoint list as it would evolve) and asks the library (llama_lm_prefill_snap_request) for the recurrent state at the token

  --lm-prefill-prearm MODE     off | always | auto: when the slots go idle after a request, prepare the pass of the next one (llama_lm_prefill_arm with seq_id -1); auto = when at least 2 of the last 4 requests used the pass

def patch(path, edits):
    d = open(path, "rb").read().decode("utf-8")
    crlf = "\r\n" in d
    text = d.replace("\r\n", "\n")
    for old, new in edits:
        if text.count(old) != 1:
            sys.exit("anchor matched %d times in %s: %r" % (text.count(old), path, old[:100]))
        text = text.replace(old, new)
    if os.environ.get("PATCH_DRY"):
        print("DRY-OK", path)
        return
    if os.environ.get("G22_APPLY") != "1":
        sys.exit("refusing to edit the nested tree: set G22_APPLY=1 (PATCH_DRY=1 first)")
    if crlf:
        text = text.replace("\n", "\r\n")
    open(path, "wb").write(text.encode("utf-8"))
    print("patched", path)
